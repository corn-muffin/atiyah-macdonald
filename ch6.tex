\begin{enumerate}
	\item[6.1.] {
      \begin{enumerate}
        \item Let $M$ be a Noetherian $A$-module and $u:M\to M$ a module homomorphism.
          If $u$ is surjective, then it is an isomorphism.
        \item If $M$ is Artinian and $u$ is injective, then again $u$ is an isomorphism.
      \end{enumerate}
    }
\end{enumerate}

\begin{proof}
  (a) Consider the filtration $\ker u\subset \ker u^2\subset\cdots$, and the associated complex of modules
  \[
    %\cdots \to\frac{\ker u^n}{\ker u^{n+1}}\to 
    \cdots \to \frac{\ker u^2}{\ker u} \xrightarrow{~~u~~} \ker u \to 0.
  \] 
  Note that $u:M\to M$ is surjective $u^{-1}(\ker u^k)= \ker u^{k+1}$, so each of the above maps is surjective.
  It follows by induction that the complex is zero if and only if it is bounded.
  $M$ Noetherian gives us $n>0$ such that $\ker u^n = \ker u^{n+1}$, whence the complex is bounded, and so $\ker u=0$ as desired.

  (b) Consider the filtration $\cdots \subset \im u^2\subset \im u$, and the associated complex of modules
  \[
    %\cdots \to\frac{\ker u^n}{\ker u^{n+1}}\to 
    0\to \cok u \xrightarrow{~~u~~} \frac{\im u}{\im u^2} \to \cdots
  \] 
  The kernel of each map is of the form $u^{-1}(\im u^{k+1})= u^{-1}(u(\im u^k)))=\im u^k$ by injectivity of $u$, so each map is injective.
  It follows by induction that the complex is zero if and only if it is bounded.
  $M$ Artinian gives us $n> 0$ such that $\im u^n = \im u^{n+1}$, whence the complex is bounded, and so $\cok u=0$ as desired.
\end{proof}

\begin{enumerate}
  \item[6.2] {
    Let $M$ be an $A$-module.
    If every non-empty set of finitely generated submodules of $M$ has a maximal element, then $M$ is Noetherian.
  }
\end{enumerate}

\begin{proof}
  Let $\Sigma$ be a non-empty set of submodules of $M$, and let $\Sigma'$ be the set of finitely generated submodules of elements of $\Sigma$.
  Then there is a maximal element $N'\in \Sigma'$, which is in particular finitely generated and contained in some $N\in \Sigma$.
  %If $N'\subsetneq N$, then we can take $x\in N\setminus N'$, whence $N'\subset (N',x)\subset N$ violates maximality of $N'$.
  It suffices to show that $N$ is a maximal element of $\Sigma$.
  
  Suppose that $N\subsetneq P\in \Sigma$, and take some $x\in P\setminus N$.
  Then $N'\subsetneq(N',x)\subset P$ is contained in $\Sigma'$, contradicting maximality of $N'$.
  Hence, $N\in \Sigma$ is maximal, whence $M$ is Noetherian.
\end{proof}

\begin{enumerate}
	\item[6.4.] {
      Let $M$ be a Noetherian $A$-module, and let $\a$ be the annihilator of $M$ in $A$.
      Prove that $A/\a$ is a Noetherian ring.

      If we replace \emph{Noetherian} by \emph{Artinian}, is this result still true?
    }
\end{enumerate}

\begin{proof}
  $M$ Noetherian gives us generators $x_1,\ldots,x_n$ of $M$, so that $\a = \bigcap_{i=1}^n \Ann(x_i)$.
  Further, $M^{\oplus n}$ is Noetherian by 6.46
  Then we have an $A$-module morphism
  \begin{align*}
    \phi: A~ &\to M^{\oplus n} \\
    r ~&\mapsto (r\cdot x_i)_{i=1}^n
  \end{align*}
  with $\ker \phi=\a$, so that $A/\a\cong \im\phi\leqs M^{\oplus n}$ is a Noetherian $A$-module.
  By (6.6), $A/\a$ is a Noetherian ring, as desired.

  The result does not hold if we replace Noetherian by Artinian.
  Consider the case of $A=\ZZ$ and $M=G$ the subgroup of $\QQ/\ZZ$ consisting of elements whose order is a power of $p$, where $p$ is a fixed prime.
  Then $G$ is Artinian, as it has only the subgroups $G_0\subset G_1\subset \cdots G_n\subset\cdots$ of order $p^n$ for each $n\geqs 0$.
  However $\a = \bigcap_{n\geqs 0} p^n\ZZ=(0)$, so that $A/\a\cong \ZZ/(0)\cong \ZZ$ is not Artinian by considering the chain $(p^n)$ for $n\geqs 0$.
\end{proof}
