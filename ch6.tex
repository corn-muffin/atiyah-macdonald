\begin{enumerate}
	\item[6.1.] {
      \begin{enumerate}
        \item Let $M$ be a Noetherian $A$-module and $u:M\to M$ a module homomorphism.
          If $u$ is surjective, then it is an isomorphism.
        \item If $M$ is Artinian and $u$ is injective, then again $u$ is an isomorphism.
      \end{enumerate}
    }
\end{enumerate}

\begin{proof}
  (a) Consider the filtration $\ker u\subset \ker u^2\subset\cdots$, and the associated complex of modules
  \[
    %\cdots \to\frac{\ker u^n}{\ker u^{n+1}}\to 
    \cdots \to \frac{\ker u^2}{\ker u} \xrightarrow{~~u~~} \ker u \to 0.
  \] 
  Note that $u:M\to M$ is surjective $u^{-1}(\ker u^k)= \ker u^{k+1}$, so each of the above maps is surjective.
  It follows by induction that the complex is zero if and only if it is bounded.
  $M$ Noetherian gives us $n>0$ such that $\ker u^n = \ker u^{n+1}$, whence the complex is bounded, and so $\ker u=0$ as desired.

  (b) Consider the filtration $\cdots \subset \im u^2\subset \im u$, and the associated complex of modules
  \[
    %\cdots \to\frac{\ker u^n}{\ker u^{n+1}}\to 
    0\to \cok u \xrightarrow{~~u~~} \frac{\im u}{\im u^2} \to \cdots
  \] 
  The kernel of each map is of the form $u^{-1}(\im u^{k+1})= u^{-1}(u(\im u^k)))=\im u^k$ by injectivity of $u$, so each map is injective.
  It follows by induction that the complex is zero if and only if it is bounded.
  $M$ Artinian gives us $n> 0$ such that $\im u^n = \im u^{n+1}$, whence the complex is bounded, and so $\cok u=0$ as desired.
\end{proof}
