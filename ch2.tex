\begin{enumerate}
	\item[1.] {
      Show that $(\ZZ/m\ZZ)\otimes_\ZZ(\ZZ/n\ZZ)\cong 0$ if $m,n$ are coprime.
    }
\end{enumerate}

\begin{proof}
  If $m,n$ are coprime, then choose $a,b$ such that $am+bn= 1$.
  Then
  \begin{align*}
    1\otimes 1 &= (am+bn)\otimes (am+bn) \\
               &= bn\otimes am \\
               &= bnm\otimes a \\
               &= 0.
  \end{align*}
\end{proof}

\begin{enumerate}
	\item[2.] {
      Let $\a\lideal A$ and $M$ an $A$-module.
      Show that $(A/\a)\otimes_A M\cong M/\a M$.
    }
\end{enumerate}

\begin{proof}
  We have exact sequences
  \begin{align*}
    0\to &\a\to A\to A/\a\to 0 \\
         &\a M\to M\to (A/\a)\otimes_A M\to 0,
  \end{align*}
  but $\a M\to M$ is injective, whence we conclude our result.
\end{proof}
