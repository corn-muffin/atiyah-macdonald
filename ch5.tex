\begin{enumerate}
	\item[5.1.] {
      Let $f:A\to B$ be an integral homomorphism of rings.
      Show that $f^*:\Spec B\to \Spec A$ is a closed mapping. (This is a geometric equivalent of (5.10).)
    }
\end{enumerate}

\begin{proof}
  Take $\b\lideal B$, and recall that $f^*(V(\b))\subset \ol {f^*(V(\b))}= V(\b^c)$ by (Ex. 1.21iii).
  Now, suppose $\p\in V(\b^c)$ is a prime ideal of $A$, i.e. $\p\supset \b^c=f^{-1}(\b)$.
  Then in particular $\p\supset \ker f$, so $\p$ decends to a prime ideal $f(\p)$ of $A/\ker f\cong \im f\subset B$.

  $\im f\subset B$ is integral, so (5.10) gives us a prime $\q\in \Spec B$ with $\q\cap \im f = f(\p)$.
  We have that $\q\supset f(\p) \supset f(\b^c)$ implies $\q\supset \b^{ce}=\b$, i.e. $\q\in V(\b)$.
  Further,
  \begin{align*}
    f^*(\q) &= f^{-1}(\q) \\
            &= f^{-1}(\q\cap \im f) \\
            &= f^{-1}(f(\p)) \\
            &= \p,
  \end{align*}
  since $\p\supset \ker f$.
  Thus, $\p\in f^*(V(\b))$, so in fact $f^*(V(\b))=\ol {f^*(V(\b))}$ is closed.
\end{proof}

\begin{enumerate}
	\item[5.2.] {
      Let $B$ be integral over $A\subset B$, and let $f:A\to \Omega$ be a homomorphism of $A$ into an algebraically closed field $\Omega$.
      Show that $f$ can be extended to a homomorphism of $B$ into $\Omega$.
    }
\end{enumerate}

\begin{proof}
  $f:A\to \Omega$ corresponds to a maximal ideal $\p=\ker f\lideal A$ and an inclusion $\ol f:A/\p\hookrightarrow \Omega$.
  (5.10) supplies us with a prime ideal $\q\lideal B$ such that $\q\cap A= \p$, which is in fact maximal by (5.8).
  Then $\pi:B\to B/\q$ restricts to a map 
  \[
    \pi|_A:A\to \frac{A}{\q\cap A}=\frac{A}\p\subset \frac{B}\q.
  \] 
  (5.6) gives us that $K:=B/\q$ is integral over $k:=A/\p$.

  Let $(k,\ol f)\leqs (C,\ol g)$ be a maximal element of the poset $\Sigma$ of maps from subrings of $K$ to $\Sigma$, ordered by restriction.
  In particular, it is a valuation ring of $K$, so $K\supset C\supset \ol k=K$ by (5.22), whence $K=C$.
  Then $g:=\ol g\circ \pi:B\to \Omega$ has 
  \[
    g|_A=\ol g|_k\circ \pi|_A= \ol f\circ \pi|_A=f.
  \]
\end{proof}

\begin{enumerate}
	\item[5.3.] {
      Let $f:B\to B'$ be a morphism of $A$-algebras, and let $C$ be an $A$-algbera.
      If $f$ is integral, prove that $f\otimes 1:B\otimes_A C\to B'\otimes C$ is integral.
    }
\end{enumerate}

\begin{proof}
  We have that $B'$ is integral over $f(B)\leqs B'$.
  That is, $f(B)[b_1',\ldots, b_n']$ is finitely generated for all $b_1',\ldots, b_n'\in B$.
  Note as well that $(f\otimes 1)(B\otimes C)\cong f(B)\otimes C$.
  Then, for arbitrary $\sum_{i=1}^n b_i'\otimes c_i\in B'\otimes_A C$,
  \begin{align*}
    (f\otimes 1)(B\otimes_A C)[\sum b_i'\otimes c_i]
    %(f\otimes 1)(B\otimes_A C)[b_1'\otimes c_1+\cdots+ b_n'\otimes c_n]
    &\leqs f(B)[b_1',\ldots, b_n']\otimes_A C[c_1,\ldots, c_n].
  \end{align*}
  The inclusions
  \begin{align*}
    f(B)\otimes_A C\inj f(B)[b_1',\ldots, b_n'] \otimes_A C\inj f(B)[b_1',\ldots, b_n']\otimes_A C[c_1,\ldots, c_n]
  \end{align*}
  are finite by closure of finiteness under base extension (2.17), so the composition is finite by closure of finiteness under composition (2.16).
  Then our arbitrary element of $B'\otimes_A C$ is integral over $f(B)\otimes_A C$ by (5.1iii), so that $f\otimes 1$ is integral.
\end{proof}

\begin{enumerate}
  \item[5.4] {
      Let $A\subset B$ be a subring such that $B$ integral over $A$.
      Let $\n$ be a maximal ideal of $B$ and let $\m=\n\cap A$ be the corresponding maximal ideal of $A$.
      Is $B_\n$ necessarily integral over $A_\m$?
    }
\end{enumerate}

\begin{proof}
  We will show that this need not hold in general.
  Consider $A:=k[x^2-1]\subset k[x]=$.
  Clearly $k[x]$ is integral over $A$, as it is a finitely generated $A$ module.
  Consider the maximal ideal $\n=(x-1)$.
  We have that $\n\supset (x^2-1)$ and $\m:=(x^2-1)^c$ is maximal in $A$, so $\n\cap A\supset \m$ with $\n$ a proper ideal implies $\n\cap A=\m$.

  Now, suppose that $1/(x+1)\in k[x]_{(x-1)}$ is integral over $A_\m$.
  Then we have $n\geqs0$ and $a_0,\ldots,a_n\in A_\m$ such that $(x+1)^{-n-1}=a_n(x+1)^{-n}+\cdots+a_0$.
  Multiplying by $(x+1)^{n+1}$, we get
  \begin{align*}
    1 = a_n (x+1) +\cdots + a_0 (x+1)^{n+1}\in (x+1)k[x]_{(x+1)},
  \end{align*}
  since $(x+1)\cap k[x^2-1]=(x^2-1)$ by the same argument as above.

  Note that $(x+1)$ is exactly the kernel of the evaluation map $f:k[x]\to k:x\mapsto -1$.
  Then $k[x]_{(x+1)}=k[x]_S$ where $S= k[x]-\ker f$, so that $f(S)\subset k^\times$ gives us $k_{f(S)}\cong k$.
  Hence, we receive a map of localizations
  \[
    \begin{tikzcd}
      k[x]_{(x+1)}\ar[r,twoheadrightarrow,"f"] & k,
    \end{tikzcd}
  \] 
  under which $1\mapsto 1$ and $a_n(x+1)+\cdots+a_0(x+1)^{n+1}\mapsto 0$.
  $1\neq 0$ in $k$, so this contradicts our assumption.
\end{proof}

\begin{enumerate}
  \item[5.7] {
      Let $A$ be a subring of $B$ such that $B- A$ is closed under multiplication. 
      Show that $A$ is integrally closed in $B$.
    }
\end{enumerate}

\begin{proof}
  Suppose that $b\in B-A$ is integral over $A$, so that $a_0=a_1b+\cdots +a_nb^n+b^{n+1}$ for $a_i\in A$ and minimal $n\geqs 0$.
  In fact, we may assume $n>0$, as otherwise $b\in A$.
  We have
  \begin{align*}
    a_0 = b(a_1+\cdots + a_nb^{n-1}+b^n) \in A
  \end{align*}
  implies $a_1+\cdots + a_nb^{n-1}+b^n=a\in A$, as otherwise $a_0\in B-A$ by closure.
  Then $a-a_1= a_2b+\cdots +a_nb^{n-1}+b^n$ is of strictly lower degree, contradicting minimality of $n$.
\end{proof}

\begin{enumerate}
  \item[5.30] {
      Let $A$ be a valuation ring of a field $K$.
      The group $A^\times$ of units of $A$ is a subgroup of the multiplicative group $K^\times$ of $K$.

      Let $\Gamma = K^\times/A^\times$.
      If $\xi,\eta\in \Gamma$ are represented by $x,y\in K$, define $\xi\geqs \eta$ to mean $xy^{-1}\in A$.
      Show that this defines a total ordering on $\Gamma$ which is compatible with the group structure (i.e., $\xi\geqs \eta\implies \xi\omega\geqs \eta\omega$ for all $\omega\in \Gamma$).
      In other words, $\Gamma$ is a totally ordered abelian group.
      It is called the \emph{value group of $A$}.

      Let $v:K^\times\to \Gamma$ be the canonical homomorphism.
      Show that $v(x+y)\geqs \min(v(x), v(y))$ for all $x,y\in K^\times$.
    }
\end{enumerate}

\begin{proof}
  Clearly $\geqs$ is reflexive and transitive.
  If $\xi\geqs \eta$ and $\eta\geqs \xi$, then $xy^{-1}\in A$ and $yx^{-1}\in A$ implies $xy^{-1}\in A^\times$, so that $\xi=\eta$ gives us anti-symmetry.
  It is total by definition of a valuation ring.
  Now, we show it is compatible with the group structure.
  Let $\xi\geqs \eta$, $\omega\in \Gamma$, and take representatives $x,y,z\in K$.
  Then $(xz)(yz)^{-1}=xy^{-1}\in A$ implies $\xi\omega\geqs \eta\omega$.

  Suppose, without loss of generality, that $v(x)\geqs v(y)$, i.e. $xy^{-1}\in A$.
  Then 
  \[
    (x+y)y^{-1}=xy^{-1}+1\in A
  \]
  implies $v(x + y)\geqs v(y)$, as desired.
  Then $d(x,y):= -v(x-y)$ defines an ultrametric on $K$, thus giving $K$ the structure of a non-archimedean field.
\end{proof}

\begin{enumerate}
  \item[5.31] {
      Conversely, let $\Gamma$ be a totally ordered abelian group (written \emph{additively}), and let $K$ be a field.
      A \emph{valuation of $K$ with values in $\Gamma$} is a mapping $v:K^\times \to \Gamma$ such that
      \begin{enumerate}
        \item $v(xy)= v(x)+v(y)$,
        \item $v(x+y)\geqs \min(v(x),v(y))$,
      \end{enumerate}
      for all $x,y\in K^\times$.
      Show that the set of elements $x\in K$ such that $x=0$ or $v(x)\geqs 0$ is a valuation ring of $K$.
      This ring is called the valuation ring of $v$, and the subgroup $v(K^\times)$ is the \emph{value group} of $v$.

      Thus, the concepts of valuation ring and valuation are essentially equivalent.
    }
\end{enumerate}

\begin{proof}
  Define $A:=\{x\in K:x=0\text{ or }v(x)\geqs 0\}$.
  Then $0\in A$, $A$ is additive by (b), and multiplicative by (a).
  If $x\in K^\times$, then 
  \[
    v(x)+v(x^{-1}) = v(1)=0
  \]
  implies either $v(x)\geqs 0$ or $v(x^{-1})\geqs 0$.
  That is, $x\in A$ or $x^{-1}\in A$.
  We conclude that $A$ is indeed a valuation ring of $K$.
\end{proof}

      Let $\Gamma$ be a totally ordered abelian group.

      If $\p$ is a prime ideal of $A$, show that $v(A\setminus \p)$ is the set of elements $\geqs 0$ in an isolated subgroup $\Delta\leqs \Gamma$, and that the mapping so defined of $\Spec A$ into isolated subgroups of $\Gamma$ is surjective.

      If $p\lideal A$ is prime, what are the value groups of the valuation rings $A/\p$, $A_\p$?
    }
\end{enumerate}

\begin{proof}
  Suppose $\alpha=v(a)\in v(A\setminus \p)$, and let $0\leqs \beta\leqs \alpha$ for $\beta=v(b)\in v(A)$.
  Then $ab^{-1}\in A$, so $b\in A\setminus \p$, as otherwise $a=(ab^{-1})b\in \p$.
  Hence, $\beta\in v(A\setminus \p)$.

  Now, let $\Delta\leqs \Gamma$ be an isolated subgroup, and consider $\p:=v^{-1} (\Gamma\setminus\Delta)\cup\{0\}$.
  Additivity is clear.
  If $b\in v^{-1}(\Gamma\setminus\Delta)$ and $a\in A\setminus\{0\}$, then $v(ab)=v(a)+v(b)\geqs v(b)\geqs 0$, so $v(ab)\notin \Delta$ gives us that $\p$ is an ideal.
  Finally, if $a,b\in A\setminus \p$, then $v(ab)=v(a)+v(b)\in \Delta$ implies $ab\in A\setminus \p$, whence $\p$ is prime.
  It is clear that $\p\mapsto \Gamma$ under the mapping of $\Spec A$.

  The value group of $A_\p$ is
  \begin{align*}
    \frac{K^\times}{A_\p^\times}
    &= \frac{K^\times}{A_\p\setminus \p A_\p} \\
    &= \frac{K^\times/A^\times}{(A_\p\setminus \p A_\p)/A^\times} \\
    &= \frac{\Gamma}{\pm v(A\setminus \p)} \\
    &= \Gamma/\Delta.
  \end{align*}

\begin{enumerate}
  \item[5.33] {
      Let $\Gamma$ be a totally ordered abelian group, $k$ any field, and $A:=k[\Gamma]$ the group algebra of $\Gamma$ over $k$.
      Show that $A$ is an integral domain.

      If $u=\lambda_1\alpha_1+\cdots+\lambda_n\alpha_n$ is any nonzero element of $A$, where each $\lambda_i\neq0$ and $\alpha_1<\cdots<\alpha_n$, define $v_0(u):=\alpha_1$.\footnote{This arises from the natural $\Gamma$-graded structure of $k[\Gamma]$.}
      Show that $v_0:A\setminus\{0\}\to \Gamma$ satisfies conditions (a) and (b) of (Ex. 5.31).
      
      Let $K$ be the fraction field of $A$.
      Show that $v_0$ can be uniquely extended to a valuation $v$ of $K$, and that the value group of $v$ is precisely $\Gamma$.
    }
\end{enumerate}

\begin{proof}
  Take $u,u\in A$ nonzero and write
  \begin{align*}
    u &=\lambda_1x_{\alpha_1}+\cdots +\lambda_mx_{\alpha_m}, \\
    u'&=\lambda_1'x_{\alpha_1'}+\cdots +\lambda_n'x_{\alpha_n'}
  \end{align*}
  for $\lambda_i,\lambda_i'\neq 0$ and $\alpha_1<\cdots \alpha_m$, $\alpha_1'<\cdots \alpha_n'$.
  Suppose by way of contradiction that
  \begin{align*}
    0&= uu' \\
     &= \sum_{i=1}^m\sum_{j=1}^n \lambda_i\lambda'_j x_{\alpha_i+\alpha_j'}.
  \end{align*}
  Then $\alpha_1+\alpha_1'<\alpha_i+\alpha_j'$ for all $i,j\neq 1$, so $\lambda_1\lambda_1'x_{\alpha_1+\alpha_1'}=0$ implies $\lambda_1=0$ or $\lambda_1'=0$, a contradiction.

  Let $u,u'\in A$ nonzero. 
  By the above, $uu'=\lambda_1\lambda_1'x_{\alpha_1+\alpha_1'}+\cdots$, so $v_0(uu')=\alpha_1+\alpha_2$ gives us (a).
  For (b), we allow $m,n=0$ and induct on $k=\#\{i\mid \lambda_i+\lambda_i'=0, \alpha_i=\alpha_i'\}$.
  If $k=0$, then clearly $v_0(u+u')=\min\{v_0(u),v_0(u')\}$.
  If $k=m=n$, then $u+u'=0$, so the result holds trivially.
  If only one of $m,n=0$, then without loss of generality $n=0$, so $v_0(u+u')=v_0(u)=\min(v_0(u),\infty)$.
  Otherwise, $v_0(u+u')=v_0(u''+u''')$ for $u''=u-\lambda_1x_{\alpha_1}$, $u'''=u'-\lambda_1x_{\alpha_1}$, and $k'=k-1$, so
  \begin{align*}
    v_0(u+u') &= v_0(u''+u''') \\
            &\geqs \min(v_0(u''),v_0(u''')) \\
            &\geqs \min(v_0(u),v_0(u')),
  \end{align*}
  whence the result follows by induction.

  Take $K=\Frac A$.
  Then for any $x,y\in A$ and $y\neq 0$, 
  \[
    v(x) = v((xy^{-1})y)= v(xy^{-1})+v(y)
  \]
  forces $v(xy^{-1})=v_0(x)-v_0(y)$.
  Clearly this is a valuation on $K$, and $v:K^\times\to \Gamma$ is surjective by construction, so $\Gamma$ is the valuation group of $v$.
\end{proof}

    \p
    &= (g\circ f)^{-1}(\p A_\p) \\
    &= f^{-1}(g^{-1}(\p A_\p)) \\
    &
  \end{align*}

  \begin{align*}
    g(\n)=(g\circ f)(\m)=\m
  \end{align*}
\end{proof}
