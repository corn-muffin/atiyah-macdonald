	\item[2.] {
      Let $A$ be a ring and let $A[x]$ be the univariate polynomial ring of $A$.
      Let $f=a_0 + a_1x + \cdots + a_nx^n\in A[x]$.
      Prove that
      \begin{enumerate}
        \item $f$ is a unit in $A[x]$ $\iff$ $a_0$ is a unit in $A$ and $a_1,\ldots, a_n$ are nilpotent.
        \item $f$ is nilpotent $\iff a_0,a_1,\ldots, a_n$ are nilpotent.
        \item $f$ is a zero divisor $\iff$ there exists $a\neq 0$ in $A$ such that $af=0$.
        \item $f$ is said to be \emph{primitive} if $(a_0,a_1,\ldots,a_n)=(1)$.
          Prove that if $f,g\in A[X]$, then $fg$ is primitive $\iff$ $f$ and $g$ are primitive.
      \end{enumerate}
    }
\end{enumerate}

\begin{enumerate}
	\item[5.] {
      Let $A$ be a ring and $A[[x]]$ the ring of formral power series $f=\sum_{n=0}^\infty a_n x^n$ with coefficients in $A$.
      Show that
      \begin{enumerate}
        \item $f$ is a unit in $A[[x]]$ $\iff$ $a_0$ is a unit in $A$.
        \item If $f$ is nilpotent, then $a_n$ is nilpotent for all $n\geqs 0$. Is the converse true?
        \item $f$ belongs to the Jacobson radical of $A[[x]]\iff a_0$ belongs to the Jacobson radical of $A$.
        \item The contraction of a maximal ideal $m$ of $A[[x]]$ is a maximal ideal of $A$, and $m$ is generated by $m^c$ and $x$.
        \item Every prime ideal of $A$ is the contraction of a prime ideal of $A[[x]]$.
      \end{enumerate}
    }
\end{enumerate}

\begin{proof}

      \begin{enumerate}
        \item if $\a$ is the ideal generated by $E$,

          \[
            V \left(\bigcup_{i\in I} E_i\right)= \bigcap_{i\in I}V(E_i).
          \] 
        \item $V(\a\cap \b)=V(\a\b)= V(\a)\cap V(\b)$ for an ideals $\a,\b\trianglelefteq A$.
      \end{enumerate}
    }
