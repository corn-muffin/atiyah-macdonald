      Show that $\Sigma$ has maximal elements, and that the maximal elements of $\Sigma$ are prime ideals.

\begin{enumerate}
	\item[7.5.] {
      Let $A$ be a Noetherian ring, $B$ a finitely generated $A$-algebra, $G$ a finite group of $A$-automorphism of $B$, and $B^G$ the set of all elements of $B$ which are left fixed by every element of $G$.
      Show that $B^G$ is a finitely generated $A$-algebra.
    }
\end{enumerate}

\begin{enumerate}
	\item[7.15.] {
      Let $A$ be a Noetherian local ring, $\m$ its maximal ideal, and $k$ its residue field, and let $M$ be a finitely generated $A$-module.
      Then the following are equivalent:
      \begin{enumerate}
        \item $M$ is free;
        \item $M$ is flat;
        \item the mapping $\m\otimes M\to A\otimes M$ is injective;
        \item $\Tor^A_I(k,M)=0$.
      \end{enumerate}
    }
\end{enumerate}

\begin{enumerate}
	\item[7.16.] {
      Let $A$ be a Noetherian local ring, and let $M$ be a finitely generated $A$-module.
      Then the following are equivalent:
      \begin{enumerate}
        \item $M$ is a flat $A$-mdoule;
        \item $M_\p$ is a free $A_\p$-module for all $\p\in \Spec A$;
        \item $M_\m$ is a free $A_\m$-module, for all maximal ideal $\m$.
      \end{enumerate}
      In other words, flat = locally free.
    }
\end{enumerate}
