\begin{enumerate}
	\item[10.3.] {
      Let $A$ be a Noetherian ring, $\a$ an ideal, and $M$ a finitely generated $A$-module.
      Using Krull's Theorem and (Ex. 3.14), prove that
      \begin{align*}
        \bigcap_{n=1}^\infty \a^n M = \bigcap_{\m\supset\a}\ker(M\to M_\m),
      \end{align*}
      where $\m$ runs over all maximal ideals containing $\a$.

      Deduce that
      \[
        \widehat M=0\iff \Supp(M)\cap V(\a)=\0.
      \] 
    }
\end{enumerate}

\begin{proof}
  By Krull's Theorem, it suffices to show that $x\in \bigcap_{\m\supset \a}\ker(M\to M_\m)$ if and only if $x$ is annihilated by some element of $1+\a$.
  We have $x\in \ker(M\to M_\m)$ if and only if $(x)_\m=0$.
  Then, by (Ex. 3.14), $x\in \bigcap_{\m\supset \a}\ker(M\to M_\m)$, implies $\a=\a(x)$.
  (2.5) gives us that $x$ is annihilated by some element of $1+\a$, as desired.

  Conversely, suppose there is $\alpha\in \a$ with $(1+\alpha)x=0$.
  Then $(x)/\a(x)=0$ implies $(x)_\m/\a_\m(x)\m=0$ for maximal ideals $\m\supset \a$, whence $(x)_\m=0$ for all $\m\supset \a$ by Nakayama's lemma.
  Thus, $x\in \bigcap_{\m\supset \a}\ker(M\to M_\m)$, as desired.

  By the above,
  \begin{align*}
    \ker (M\to \widehat M)
      &= \bigcap_{n=1}^\infty \a^n M \\
      &= \bigcap_{\m\supset \a}\ker(M\to M_\m).
  \end{align*}
  For any $\p\in V(\a)$, we have $\a\subset \p\subset \m$ for some $\m$ maximal.
  Then $A-\m\subset A -\p$ gives us
  \[
    \ker(M\to M_\m\to M_\p) \supset \ker(M\to M_\m),
  \] 
  whence
  \begin{align*}
    \ker(M\to \widehat M)
      &=\bigcap_{\p\subset \a}\ker(M\to M_\p).
  \end{align*}
  Thus, $\Supp(M)\cap V(\a)=\0$ if and only if $M=\ker(M\to \widehat M)$.
  Clearly $\widehat M=0$ is a sufficient condition.
  Conversely, if every constant sequence converges to 0, then every every sequence converges to 0, i.e. $\widehat M=0$.
\end{proof}

	\item[10.6.] {
      Let $A$ be a Noetherian ring and $\a$ an ideal in $A$.
      Prove that $\a$ is contained in the Jacobson radical of $A$ if and only if every maximal ideal of $A$ is closed for the $\a$-topology.
